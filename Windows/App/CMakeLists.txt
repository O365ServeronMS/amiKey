# amiKey.exe: tray app + hooks. Built only on Windows (MSVC).
# One version for every platform: project.yml's MARKETING_VERSION, unless -DAMIKEY_VERSION overrides it.
unset(AMIKEY_VERSION CACHE)   # never a stale cached value
if(NOT AMIKEY_VERSION)
  file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/../../project.yml" _mv REGEX "MARKETING_VERSION: \"")
  string(REGEX REPLACE ".*MARKETING_VERSION: \"([^\"]+)\".*" "\\1" AMIKEY_VERSION "${_mv}")
endif()
if(NOT AMIKEY_BUILD)
  file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/../../project.yml" _bv REGEX "CURRENT_PROJECT_VERSION: \"")
  string(REGEX REPLACE ".*CURRENT_PROJECT_VERSION: \"([^\"]+)\".*" "\\1" AMIKEY_BUILD "${_bv}")
endif()
message(STATUS "amiKey version ${AMIKEY_VERSION} (${AMIKEY_BUILD})")
# "1.0.32" -> "1,0,32,0" for the binary FILEVERSION/PRODUCTVERSION in amiKey.rc
string(REPLACE "." "," AMIKEY_VERSION_NUM "${AMIKEY_VERSION}")
string(REGEX MATCHALL "," _commas "${AMIKEY_VERSION_NUM}")
list(LENGTH _commas _n)
while(_n LESS 3)
  string(APPEND AMIKEY_VERSION_NUM ",0")
  math(EXPR _n "${_n} + 1")
endwhile()
add_executable(amikey WIN32 main.cpp Core.cpp Prefs.cpp Site.cpp Settings.cpp Emoji.cpp Updater.cpp CrashReporter.cpp third_party/tweetnacl.c amiKey.rc)
set_source_files_properties(third_party/tweetnacl.c PROPERTIES COMPILE_FLAGS "/w")
set_target_properties(amikey PROPERTIES OUTPUT_NAME amiKey)
target_include_directories(amikey PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(amikey PRIVATE UNICODE _UNICODE AMIKEY_VERSION="${AMIKEY_VERSION}" AMIKEY_VERSION_NUM=${AMIKEY_VERSION_NUM} AMIKEY_BUILD="${AMIKEY_BUILD}")
target_link_libraries(amikey PRIVATE amikey_engine user32 shell32 gdi32 advapi32 imm32 psapi ole32 oleaut32 comctl32 gdiplus winhttp winmm d2d1 dwrite dwmapi normaliz)
target_link_options(amikey PRIVATE /MANIFEST:NO)
# Live test runner: a window with an edit control, types through SendInput while amiKey runs, reads the text back.
add_executable(amikey_livetest ../Tests/livetest.cpp Prefs.cpp)
target_include_directories(amikey_livetest PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(amikey_livetest PRIVATE amikey_engine user32 advapi32 shell32)
target_compile_definitions(amikey_livetest PRIVATE UNICODE _UNICODE)
